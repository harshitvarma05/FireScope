cmake_minimum_required(VERSION 3.24)
project(FireScope VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FIRESCOPE_BUILD_GUI "Build OpenGL/ImGui application" ON)
option(FIRESCOPE_ENABLE_CUDA "Enable CUDA backend when implemented and available" ON)
option(FIRESCOPE_ENABLE_METAL "Enable Metal backend when implemented and available" ON)
option(FIRESCOPE_ENABLE_OPENMP "Enable optional OpenMP backend" OFF)

include(CTest)
add_library(firescope_core src/core/TerrainGenerator.cpp src/core/Scenario.cpp src/backends/cpu/CPUBackend.cpp src/core/BackendFactory.cpp)
target_include_directories(firescope_core PUBLIC include src/backends/cpu)
include(cmake/CudaConfig.cmake)
include(cmake/MacMetal.cmake)
add_executable(FireScope src/main.cpp)
target_link_libraries(FireScope PRIVATE firescope_core)
if(FIRESCOPE_BUILD_GUI)
    include(cmake/Dependencies.cmake)
    find_package(OpenGL REQUIRED)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS shaders/terrain.vert shaders/terrain.frag)
    file(READ ${CMAKE_CURRENT_SOURCE_DIR}/shaders/terrain.vert FIRESCOPE_TERRAIN_VERTEX)
    file(READ ${CMAKE_CURRENT_SOURCE_DIR}/shaders/terrain.frag FIRESCOPE_TERRAIN_FRAGMENT)
    foreach(kind vegetation billboard sky)
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS shaders/${kind}.vert shaders/${kind}.frag)
        file(READ ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${kind}.vert FIRESCOPE_${kind}_VERTEX)
        file(READ ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${kind}.frag FIRESCOPE_${kind}_FRAGMENT)
    endforeach()
    configure_file(cmake/ShaderSources.hpp.in generated/ShaderSources.hpp @ONLY)
    add_library(firescope_geometry src/render/Camera.cpp src/render/TerrainMesh.cpp)
    target_include_directories(firescope_geometry PUBLIC include)
    target_link_libraries(firescope_geometry PUBLIC glm::glm)
    target_sources(FireScope PRIVATE src/render/OpenGLRenderer.cpp src/render/SmokeTriangle.cpp src/ui/UI.cpp)
    target_include_directories(FireScope PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_link_libraries(FireScope PRIVATE firescope_geometry imgui OpenGL::GL)
    target_compile_definitions(FireScope PRIVATE GL_SILENCE_DEPRECATION FIRESCOPE_BUILD_GUI=1)
    if(APPLE)
        set_target_properties(FireScope PROPERTIES
            MACOSX_BUNDLE TRUE
            MACOSX_BUNDLE_GUI_IDENTIFIER "org.firescope.app"
            MACOSX_BUNDLE_BUNDLE_NAME "FireScope")
    endif()
    if(BUILD_TESTING)
        add_executable(test_terrain tests/test_terrain.cpp)
        target_link_libraries(test_terrain PRIVATE firescope_geometry firescope_core)
        add_test(NAME terrain_camera_picking COMMAND test_terrain)
    endif()
endif()
if(BUILD_TESTING)
    add_executable(test_rng tests/test_rng.cpp)
    target_link_libraries(test_rng PRIVATE firescope_core)
    add_test(NAME deterministic_rng COMMAND test_rng)
    add_executable(test_cpu_sim tests/test_cpu_sim.cpp)
    target_link_libraries(test_cpu_sim PRIVATE firescope_core)
    add_test(NAME cpu_simulation COMMAND test_cpu_sim)
    add_test(NAME headless_cli COMMAND FireScope --backend cpu --grid 128 --steps 30 --seed 42 --headless)
    add_executable(test_backend_consistency tests/test_backend_consistency.cpp)
    target_link_libraries(test_backend_consistency PRIVATE firescope_core)
    add_test(NAME backend_consistency COMMAND test_backend_consistency)
    set_tests_properties(backend_consistency PROPERTIES SKIP_RETURN_CODE 77)
    add_executable(test_inspection tests/test_inspection.cpp)
    target_link_libraries(test_inspection PRIVATE firescope_core)
    add_test(NAME cell_inspection_model COMMAND test_inspection)
    add_executable(test_scenario tests/test_scenario.cpp)
    target_link_libraries(test_scenario PRIVATE firescope_core)
    add_test(NAME scenario_snapshot_comparison COMMAND test_scenario)
endif()
